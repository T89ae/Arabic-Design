    function generateRandomData() {
        // تحديث الأرقام الرئيسية بشكل عشوائي ولكن واقعي
        dashboardData.totalWorkers = Math.floor(10 + Math.random() * 15);
        dashboardData.currentBalance = +(dashboardData.currentBalance + (Math.random() - 0.5) * 800).toFixed(2);
        dashboardData.currentBalance = Math.max(0, dashboardData.currentBalance);
        dashboardData.totalSales = +(dashboardData.totalSales + (Math.random() - 0.5) * 1200).toFixed(2);
        dashboardData.totalSales = Math.max(0, dashboardData.totalSales);
        dashboardData.pendingTasks = Math.max(0, dashboardData.pendingTasks + Math.floor(Math.random() * 3) - 1);
        dashboardData.transfers = Math.max(0, dashboardData.transfers + Math.floor(Math.random() * 2) - 0);
        dashboardData.inventory = Math.max(0, dashboardData.inventory + Math.floor(Math.random() * 7) - 2);
        dashboardData.saudiEmployees = Math.min(dashboardData.totalWorkers, Math.max(0, dashboardData.saudiEmployees + Math.floor(Math.random() * 3) - 1));
        dashboardData.clients = Math.max(0, dashboardData.clients + Math.floor(Math.random() * 5) - 1);
        dashboardData.expenses = +(dashboardData.expenses + (Math.random() - 0.5) * 400).toFixed(2);
        dashboardData.expenses = Math.max(0, dashboardData.expenses);

        // تحديث بيانات الإيرادات الشهرية
        for (let i = 0; i < revenueData.length; i++) {
            let change = (Math.random() - 0.5) * 800;
            revenueData[i] = Math.max(5000, revenueData[i] + change);
            revenueData[i] = Math.round(revenueData[i]);
        }

        // تحديث بيانات المصروفات الشهرية
        for (let i = 0; i < expenseDataMonthly.length; i++) {
            let change = (Math.random() - 0.5) * 500;
            expenseDataMonthly[i] = Math.max(2000, expenseDataMonthly[i] + change);
            expenseDataMonthly[i] = Math.round(expenseDataMonthly[i]);
        }

        // تحديث توزيع النفقات
        for (let i = 0; i < expenseValues.length; i++) {
            let change = (Math.random() - 0.5) * 300;
            expenseValues[i] = Math.max(100, expenseValues[i] + change);
            expenseValues[i] = Math.round(expenseValues[i]);
        }

        // إضافة مهمة جديدة أحياناً
        if (Math.random() > 0.7) {
            const newTasks = [
                { title: 'مراجعة عقود جديدة', status: 'معلقة' },
                { title: 'تحديث بيانات الموظفين', status: 'قيد التنفيذ' },
                { title: 'صيانة دورية للمعدات', status: 'مستعجلة' },
                { title: 'اجتماع مع الشركاء', status: 'معلقة' }
            ];
            const randomTask = newTasks[Math.floor(Math.random() * newTasks.length)];
            const today = new Date();
            const dateStr = today.toISOString().split('T')[0];
            tasksList.unshift({
                id: Date.now(),
                title: randomTask.title,
                date: dateStr,
                status: randomTask.status
            });
            if (tasksList.length > 8) tasksList.pop();
        }
    }

    function showToast(message, isError = false) {
        const existingToast = document.querySelector('.toast-notification');
        if (existingToast) existingToast.remove();
        
        const toast = document.createElement('div');
        toast.className = 'toast-notification';
        toast.style.background = isError ? '#dc2626' : '#10b981';
        toast.innerHTML = `<i class="fas ${isError ? 'fa-exclamation-triangle' : 'fa-check-circle'}"></i> ${message}`;
        document.body.appendChild(toast);
        
        setTimeout(() => {
            toast.style.opacity = '0';
            toast.style.transform = 'translateY(20px)';
            setTimeout(() => toast.remove(), 300);
        }, 3000);
    }

    function updateUI() {
        // تحديث البطاقات مع تأثير وميض
        renderStatsCards();
        
        // تحديث الرسوم البيانية
        if (revenueExpenseChartInstance) {
            revenueExpenseChartInstance.data.datasets[0].data = [...revenueData];

    // دالة تحديث كاملة مع تأثيرات
    function performUpdate(showToastMessage = true) {
        generateRandomData();
        updateUI();
        
        if (showToastMessage) {
            const now = new Date();
            const timeStr = now.toLocaleTimeString('ar-SA', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
            showToast(`تم تحديث البيانات بنجاح ${timeStr}`);
        }
    }

        const statsGrid = document.getElementById('statsGrid');
        const statsMapping = [
            { label: 'إجمالي العمال', key: 'totalWorkers', icon: 'fas fa-users' },
            { label: 'الرصيد الحالي', key: 'currentBalance', icon: 'fas fa-wallet', isCurrency: true },
            { label: 'إجمالي المبيعات', key: 'totalSales', icon: 'fas fa-chart-simple', isCurrency: true },

    function renderTasks() {
        const container = document.getElementById('tasksListContainer');
        if (!tasksList.length) {
            container.innerHTML = '<div class="empty-state">لا توجد مهام مسجلة حالياً</div>';
            return;
        }
        let tasksHtml = '';
        tasksList.slice(0, 6).forEach(task => {
            let statusStyle = '';
            let badgeColor = '#f97316';
            if (task.status === 'مكتملة') {
                statusStyle = 'background:#e6f7e6; color:#2e7d32';
                badgeColor = '#22c55e';
            } else if (task.status === 'معلقة') {
                statusStyle = 'background:#fff3e0; color:#ed6c02';
                badgeColor = '#f97316';
            } else if (task.status === 'مستعجلة') {
                statusStyle = 'background:#ffe6e5; color:#d32f2f';
                badgeColor = '#ef4444';
            } else if (task.status === 'قيد التنفيذ') {
                statusStyle = 'background:#e3f2fd; color:#1976d2';
                badgeColor = '#3b82f6';
            }
            
            tasksHtml += `
                <div class="task-item">
                    <div class="task-info">
                        <div class="task-badge" style="background: ${badgeColor}"></div>
                        <div>
                            <div class="task-title">${task.title}</div>
                            <div class="task-date">${task.date}</div>
                        </div>
                    </div>
                    <div class="task-status" style="${statusStyle}">${task.status}</div>
                </div>
            `;
        });
        container.innerHTML = tasksHtml;
    }

    // --------------------------------------------------------------
    // دوال التحكم في التحديث التلقائي
    // --------------------------------------------------------------
    function startAutoUpdate() {
        if (updateInterval) clearInterval(updateInterval);
        if (countdownInterval) clearInterval(countdownInterval);
        
        autoUpdateEnabled = true;
        countdown = 30;
        updateCountdownDisplay();
        
        // تحديث كل 30 ثانية
        updateInterval = setInterval(() => {
            if (autoUpdateEnabled) {
                performUpdate(true);
                countdown = 30;
                updateCountdownDisplay();
            }
        }, 30000);
        
        // تحديث العداد التنازلي كل ثانية
        countdownInterval = setInterval(() => {
            if (autoUpdateEnabled && countdown > 0) {
                countdown--;
                updateCountdownDisplay();
            }
        }, 1000);
        
        updateAutoUpdateButtonUI();
    }
    
    function stopAutoUpdate() {
        autoUpdateEnabled = false;
        if (updateInterval) clearInterval(updateInterval);
        if (countdownInterval) clearInterval(countdownInterval);
        updateInterval = null;
        countdownInterval = null;
        updateAutoUpdateButtonUI();
        showToast('تم إيقاف التحديث التلقائي', false);
    }
    
    function toggleAutoUpdate() {
        if (autoUpdateEnabled) {
            stopAutoUpdate();
        } else {
            startAutoUpdate();
            showToast('تم تشغيل التحديث التلقائي - سيتم التحديث كل 30 ثانية', false);
        }
    }
    
    function updateCountdownDisplay() {
        const countdownElement = document.getElementById('countdownTimer');
        if (countdownElement) {
            countdownElement.textContent = countdown;
        }
        const icon = document.getElementById('autoUpdateIcon');
        if (icon && autoUpdateEnabled) {
            icon.style.animation = 'none';
            icon.offsetHeight;
            icon.style.animation = 'spin 1s linear infinite';
        } else if (icon) {
            icon.style.animation = 'none';
        }
    }
    
    function updateAutoUpdateButtonUI() {
        const btnIcon = document.getElementById('autoUpdateBtnIcon');
        const btnText = document.getElementById('autoUpdateBtnText');
        if (autoUpdateEnabled) {
            btnIcon.className = 'fas fa-pause';
            btnText.textContent = 'إيقاف التحديث التلقائي';
        } else {
            btnIcon.className = 'fas fa-play';
            btnText.textContent = 'تشغيل التحديث التلقائي';
        }
    }
    
    function manualUpdate() {
        if (!autoUpdateEnabled) {
            performUpdate(true);
            showToast('تم التحديث اليدوي بنجاح', false);
        } else {
            performUpdate(true);
            countdown = 30;
            updateCountdownDisplay();
            showToast('تم التحديث اليدوي - تم إعادة ضبط العداد', false);
        }
    }

    function setCurrentDate() {
